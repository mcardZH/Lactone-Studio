include("${_es_build_root}/controls/relative-paths.cmake")
add_library("mctc-lib" UNKNOWN IMPORTED GLOBAL)
set_target_properties("mctc-lib" PROPERTIES IMPORTED_LOCATION "${_es_build_root}/outputs/mctc-lib/libmctc-lib-0.dll.a" INTERFACE_INCLUDE_DIRECTORIES "${_es_build_root}/build/mctc-lib/upstream/include;${_es_build_root}/source/mctc-lib/include")
add_library("mctc-lib::mctc-lib" ALIAS "mctc-lib")
add_library("mctc-lib::mctc-lib-lib" ALIAS "mctc-lib")
add_library("toml-f" UNKNOWN IMPORTED GLOBAL)
set_target_properties("toml-f" PROPERTIES IMPORTED_LOCATION "${_es_build_root}/outputs/toml-f/libtoml-f-0.dll.a" INTERFACE_INCLUDE_DIRECTORIES "${_es_build_root}/build/toml-f/upstream/include;${_es_build_root}/source/toml-f/include")

add_library("multicharge" UNKNOWN IMPORTED GLOBAL)
set_target_properties("multicharge" PROPERTIES IMPORTED_LOCATION "${_es_build_root}/outputs/multicharge/libmulticharge-0.dll.a" INTERFACE_INCLUDE_DIRECTORIES "${_es_build_root}/build/multicharge/upstream/include;${_es_build_root}/source/multicharge/include")
add_library("multicharge::multicharge" ALIAS "multicharge")
add_library("multicharge::multicharge-lib" ALIAS "multicharge")
set(BUILD_SHARED_LIBS ON CACHE BOOL "Explicit component-only configuration" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "Explicit component-only configuration" FORCE)
set(ENABLE_TESTING OFF CACHE BOOL "Explicit component-only configuration" FORCE)
set(WITH_TESTS OFF CACHE BOOL "Explicit component-only configuration" FORCE)
set(TBLITE_WITH_TESTS OFF CACHE BOOL "Explicit component-only configuration" FORCE)
set(SY_COMPONENT_TESTS OFF CACHE BOOL "Explicit component-only configuration" FORCE)
set(WITH_API ON CACHE BOOL "Explicit component-only configuration" FORCE)
set(TBLITE_WITH_API ON CACHE BOOL "Explicit component-only configuration" FORCE)
set(WITH_OpenMP ON CACHE BOOL "Explicit component-only configuration" FORCE)
set(TBLITE_WITH_OpenMP ON CACHE BOOL "Explicit component-only configuration" FORCE)
set(MCTCLIB_WITH_OpenMP OFF CACHE BOOL "Explicit component-only configuration" FORCE)
set(WITH_ILP64 OFF CACHE BOOL "Explicit component-only configuration" FORCE)
set(WITH_API_V2 OFF CACHE BOOL "Explicit component-only configuration" FORCE)
set(WITH_BLAS OFF CACHE BOOL "Explicit component-only configuration" FORCE)
set(TBLITE_WITH_DDX OFF CACHE BOOL "Explicit component-only configuration" FORCE)
set(TBLITE_WITH_HDF5 OFF CACHE BOOL "Explicit component-only configuration" FORCE)
set(TBLITE_WITH_TREXIO OFF CACHE BOOL "Explicit component-only configuration" FORCE)

add_subdirectory("${_es_build_root}/source/dftd4" upstream)
set_target_properties("dftd4-lib" PROPERTIES ARCHIVE_OUTPUT_DIRECTORY "${_es_build_root}/outputs/dftd4" RUNTIME_OUTPUT_DIRECTORY "${_es_build_root}/outputs/dftd4")
set_target_properties("dftd4-lib" PROPERTIES PREFIX "lib" OUTPUT_NAME "dftd4-4" WINDOWS_EXPORT_ALL_SYMBOLS FALSE)
target_sources("dftd4-lib" PRIVATE "${_es_build_root}/source/dftd4/zz_sy_probe.c" "${_es_build_root}/source/dftd4/preserve.def")
target_link_libraries("dftd4-lib" PRIVATE "${_es_build_root}/link/liblapack.dll.a" "${_es_build_root}/link/libblas.dll.a")
