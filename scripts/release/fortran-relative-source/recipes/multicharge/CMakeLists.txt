add_library("mctc-lib" UNKNOWN IMPORTED GLOBAL)
set_target_properties("mctc-lib" PROPERTIES IMPORTED_LOCATION "${_es_build_root}/outputs/mctc-lib/libmctc-lib-0.dll.a" INTERFACE_INCLUDE_DIRECTORIES "${_es_build_root}/build/mctc-lib/upstream/include;${_es_build_root}/source/mctc-lib/include")

set(BUILD_SHARED_LIBS ON CACHE BOOL "Explicit component-only configuration" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "Explicit component-only configuration" FORCE)
set(ENABLE_TESTING OFF CACHE BOOL "Explicit component-only configuration" FORCE)
set(WITH_TESTS OFF CACHE BOOL "Explicit component-only configuration" FORCE)
set(TBLITE_WITH_TESTS OFF CACHE BOOL "Explicit component-only configuration" FORCE)
